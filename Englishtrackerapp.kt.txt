enum class EnglishDialect(val label: String, val shortLabel: String, val flag: String) {
    US("American English (US)", "US", "🇺🇸"),
    UK("British English (UK)", "UK", "🇬🇧")
}

enum class CefrLevel(
    val code: String,
    val title: String,
    val murphyBook: String,
    val oxfordTargetQuota: Int // Целевой норматив слов Oxford 3000/5000 для уровня
) {
    A0_A1(
        code = "A0–A1",
        title = "Beginner / Elementary",
        murphyBook = "Essential Grammar in Use (Red Murphy • Ч. 1)",
        oxfordTargetQuota = 900
    ),
    A2(
        code = "A2",
        title = "Pre-Intermediate",
        murphyBook = "Essential Grammar in Use (Red Murphy • Ч. 2)",
        oxfordTargetQuota = 850
    ),
    B1(
        code = "B1",
        title = "Intermediate",
        murphyBook = "English Grammar in Use (Blue Murphy • Ч. 1)",
        oxfordTargetQuota = 800
    ),
    B2(
        code = "B2",
        title = "Upper-Intermediate",
        murphyBook = "English Grammar in Use (Blue Murphy • Ч. 2)",
        oxfordTargetQuota = 750
    ),
    C1_C2(
        code = "C1–C2",
        title = "Advanced / Mastery",
        murphyBook = "Advanced Grammar in Use (Green Murphy / Hewings)",
        oxfordTargetQuota = 1700
    );

    companion object {
        fun fromCode(code: String): CefrLevel =
            entries.find { it.code == code } ?: A0_A1
    }
}

/**
 * Слово из словаря Oxford 3000/5000, жестко связанное с карточками Anki.
 * Правило закрытия уровня: слово считается полностью закрытым (isMastered = true)
 * ТОЛЬКО тогда, когда пользователь сдал его в Anki на «Идеально» [REQUIRED_PERFECT_STREAK] раз подряд без ошибок.
 */
@Entity(tableName = "oxford_words")
data class OxfordWordEntity(
    @PrimaryKey(autoGenerate = true) val id: Int = 0,
    val levelCode: String,          // "A0–A1", "A2", "B1", "B2", "C1–C2"
    val wordUs: String,             // Написание в American English
    val wordUk: String,             // Написание в British English
    val ipaUs: String,              // Транскрипция US (IPA)
    val ipaUk: String,              // Транскрипция UK (IPA)
    val partOfSpeech: String,       // noun, verb, adj, adv, phrasal verb
    val translationRu: String,      // Перевод на русский (показывается после ответа в Anki)
    val exampleUs: String,          // Пример предложения (US)
    val exampleUk: String,          // Пример предложения (UK)
    val exampleRu: String,          // Перевод примера на русский
    // Поля интервального повторения Anki (SM-2) и идеальной сдачи:
    val perfectStreak: Int = 0,     // Серия идеальных ответов подряд (нужно 3/3 для закрытия слова)
    val intervalDays: Int = 0,      // Текущий интервал в днях
    val easeFactor: Float = 2.5f,   // Коэффициент легкости SM-2
    val nextReviewEpochDay: Long = 0L, // День следующего показа (LocalDate.toEpochDay())
    val totalReviews: Int = 0,
    val lastResultCorrect: Boolean? = null
) {
    companion object {
        const val REQUIRED_PERFECT_STREAK = 3
    }

    val isMasteredInAnki: Boolean
        get() = perfectStreak >= REQUIRED_PERFECT_STREAK

    val masteryFraction: Float
        get() = (perfectStreak.toFloat() / REQUIRED_PERFECT_STREAK.toFloat()).coerceIn(0f, 1f)

    fun displayWord(dialect: EnglishDialect): String =
        if (dialect == EnglishDialect.US) wordUs else wordUk

    fun displayIpa(dialect: EnglishDialect): String =
        if (dialect == EnglishDialect.US) ipaUs else ipaUk

    fun displayExample(dialect: EnglishDialect): String =
        if (dialect == EnglishDialect.US) exampleUs else exampleUk
}

)

// 2. ROOM DAO И БАЗА ДАННЫХ
// ============================================================================

@Dao
interface TrackerDao {
    @Query("SELECT * FROM oxford_words ORDER BY id ASC")
    fun observeAllWords(): Flow<List<OxfordWordEntity>>

    @Query("SELECT COUNT(*) FROM oxford_words")
    suspend fun getWordCount(): Int

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertWords(words: List<OxfordWordEntity>)

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertWord(word: OxfordWordEntity)

    @Update
    suspend fun updateWord(word: OxfordWordEntity)

    @Query("SELECT * FROM murphy_units ORDER BY unitNumber ASC")
    fun observeAllUnits(): Flow<List<MurphyUnitEntity>>

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertUnits(units: List<MurphyUnitEntity>)

    @Update
    suspend fun updateUnit(unit: MurphyUnitEntity)

    @Query("SELECT * FROM user_stats WHERE id = 1")
    fun observeStats(): Flow<UserStatsEntity?>

    @Query("SELECT * FROM user_stats WHERE id = 1")
    suspend fun getStatsOnce(): UserStatsEntity?

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun saveStats(stats: UserStatsEntity)
}

@Database(
    entities = [OxfordWordEntity::class, MurphyUnitEntity::class, UserStatsEntity::class],
    version = 1,
    exportSchema = false
)
abstract class AppDatabase : RoomDatabase() {
    abstract fun trackerDao(): TrackerDao

    companion object {
        @Volatile
        private var INSTANCE: AppDatabase? = null

        fun getInstance(context: Context): AppDatabase {
            return INSTANCE ?: synchronized(this) {
                val instance = Room.databaseBuilder(
                    context.applicationContext,
                    AppDatabase::class.java,
                    "oxford_murphy_tracker.db"
                ).build()
                INSTANCE = instance
                instance
            }
        }
    }
}

enum class AnkiGrade(val label: String, val subLabel: String) {
    WRONG("Неверно", "Сброс серии • повторим сегодня"),
    GOOD("Верно", "+1 к серии • по расписанию"),

     * Вычисление прогресса по каждому уровню (A0–A1, A2, B1, B2, C1–C2).
     * ЖЕСТКОЕ ПРАВИЛО: Словарь уровня считается закрытым на 100% ТОЛЬКО тогда,
     * когда каждое слово уровня сдано в Anki идеально (perfectStreak >= 3).
     */
    val levelSummariesFlow: StateFlow<List<LevelProgressSummary>> = combine(
        wordsFlow,
        unitsFlow
    ) { allWords, allUnits ->
        CefrLevel.entries.map { level ->
            val levelUnits = allUnits.filter { it.levelCode == level.code }
            val completedUnits = levelUnits.count { it.isCompleted }
            val grammarPercent = if (levelUnits.isEmpty()) 0 else
                ((completedUnits.toFloat() / levelUnits.size) * 100f).roundToInt()

            val levelWords = allWords.filter { it.levelCode == level.code }
            val masteredWords = levelWords.count { it.isMasteredInAnki }
            val isVocabFullyClosed = levelWords.isNotEmpty() && masteredWords == levelWords.size

            // Процент словаря считается по среднему прогрессу идеальных сдач в Anki,
            // но 100% дается ТОЛЬКО когда абсолютно все слова уровня закрыты идеально (3/3)!
            val rawVocabFraction = if (levelWords.isEmpty()) 0f else
                levelWords.sumOf { it.masteryFraction.toDouble() }.toFloat() / levelWords.size
            val vocabAnkiPercent = if (isVocabFullyClosed) {
                100
            } else {
                // Не позволяем показать 100%, пока хотя бы одно слово не сдано идеально до конца
                (rawVocabFraction * 100f).roundToInt().coerceAtMost(99)
            }

            val isGrammarClosed = levelUnits.isNotEmpty() && completedUnits == levelUnits.size
            val isLevelFullyClosed = isGrammarClosed && isVocabFullyClosed

            val overallPercent = if (isLevelFullyClosed) {
                100
            } else {
                ((grammarPercent + vocabAnkiPercent) / 2.0).roundToInt().coerceAtMost(99)
            }

            LevelProgressSummary(
                level = level,
                totalUnits = levelUnits.size,
                completedUnits = completedUnits,
                grammarPercent = grammarPercent,
                totalWordsInDb = levelWords.size,
                masteredWordsInAnki = masteredWords,
                vocabAnkiPercent = vocabAnkiPercent,
                overallPercent = overallPercent,
                isVocabFullyClosed = isVocabFullyClosed,
                isLevelFullyClosed = isLevelFullyClosed
            )
        }
    }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

    /**
     * Оценка карточки в Anki с обновлением интервала SM-2 и статуса идеальной сдачи.
     */
    fun submitAnkiReview(word: OxfordWordEntity, grade: AnkiGrade) {
        viewModelScope.launch {
            val today = LocalDate.now().toEpochDay()
            val updated = when (grade) {
                AnkiGrade.WRONG -> {
                    // Ошибка сбрасывает идеальную серию до 0: слово снова открыто и требует повтора
                    word.copy(
                        perfectStreak = 0,
                        intervalDays = 0,
                        easeFactor = max(1.3f, word.easeFactor - 0.2f),
                        nextReviewEpochDay = today,
                        totalReviews = word.totalReviews + 1,
                        lastResultCorrect = false
                    )
                }
                AnkiGrade.GOOD -> {
                    val newStreak = (word.perfectStreak + 1).coerceAtMost(OxfordWordEntity.REQUIRED_PERFECT_STREAK)
                    val nextInterval = when (word.intervalDays) {
                        0 -> 1
                        1 -> 3
                        else -> (word.intervalDays * word.easeFactor).roundToInt()
                    }
                    word.copy(
                        perfectStreak = newStreak,
                        intervalDays = nextInterval,
                        nextReviewEpochDay = today + nextInterval,
                        totalReviews = word.totalReviews + 1,
                        lastResultCorrect = true
                    )
                }
                AnkiGrade.PERFECT -> {
                    val newStreak = (word.perfectStreak + 1).coerceAtMost(OxfordWordEntity.REQUIRED_PERFECT_STREAK)
                    val nextInterval = when (word.intervalDays) {
                        0 -> 2
                        1 -> 5
                        else -> (word.intervalDays * (word.easeFactor + 0.15f)).roundToInt()
                    }
                    word.copy(
                        perfectStreak = newStreak,
                        intervalDays = nextInterval,
                        easeFactor = word.easeFactor + 0.1f,
                        nextReviewEpochDay = today + nextInterval,
                        totalReviews = word.totalReviews + 1,
                        lastResultCorrect = true
                    )
                }
            }
            dao.updateWord(updated)

            val stats = dao.getStatsOnce() ?: UserStatsEntity()
            dao.saveStats(stats.copy(totalAnkiReviews = stats.totalAnkiReviews + 1))
            recordStudyActivity()
        }
    }

    fun addCustomOxfordWord(
        levelCode: String,
        wordUs: String,
        wordUk: String,
        ipaUs: String,
        ipaUk: String,
        partOfSpeech: String,
        translationRu: String,
        exampleEn: String,
        exampleRu: String
    ) {
        viewModelScope.launch {
            dao.insertWord(
                OxfordWordEntity(
                    levelCode = levelCode,
                    wordUs = wordUs.trim(),
                    wordUk = wordUk.ifBlank { wordUs }.trim(),
                    ipaUs = ipaUs.ifBlank { "/${wordUs.trim()}/" },
                    ipaUk = ipaUk.ifBlank { ipaUs.ifBlank { "/${wordUk.trim()}/" } },
                    partOfSpeech = partOfSpeech.ifBlank { "word" },
                    translationRu = translationRu.trim(),
                    exampleUs = exampleEn.trim(),
                    exampleUk = exampleEn.trim(),
                    exampleRu = exampleRu.trim()
                )
            )
        }
    }

    fun toggleTimer() {
        _isTimerRunning.value = !_isTimerRunning.value
    }

    fun finishAndSaveTimerSession() {
        viewModelScope.launch {
            _isTimerRunning.value = false
            val minutesToAdd = max(1, (_sessionSeconds.value / 60.0).roundToInt())
            _sessionSeconds.value = 0
            addStudyMinutes(minutesToAdd)
        }
    }

    fun addStudyMinutes(minutes: Int) {
        viewModelScope.launch {
            val stats = dao.getStatsOnce() ?: UserStatsEntity()
            dao.saveStats(stats.copy(totalMinutesStudied = stats.totalMinutesStudied + minutes))
            recordStudyActivity()
        }
    }

    private suspend fun checkAndUpdateDailyStreak() {
        val stats = dao.getStatsOnce() ?: return
        val today = LocalDate.now().toEpochDay()
        val diff = today - stats.lastStudyEpochDay
        if (diff > 1L) {
            // Пропущен день — стрик сбрасывается до 1 при следующем занятии
            dao.saveStats(stats.copy(currentStreakDays = 0))
        }
    }

    private suspend fun recordStudyActivity() {
        val stats = dao.getStatsOnce() ?: UserStatsEntity()
        val today = LocalDate.now().toEpochDay()
        val diff = today - stats.lastStudyEpochDay
        val newStreak = when {
            diff == 0L -> max(1, stats.currentStreakDays)
            diff == 1L -> stats.currentStreakDays + 1
            else -> 1
        }
        val newLongest = max(stats.longestStreakDays, newStreak)
        dao.saveStats(
            stats.copy(
                currentStreakDays = newStreak,
                longestStreakDays = newLongest,
                lastStudyEpochDay = today
            )
        )
    }
}

@Composable
fun EnglishTrackerRoot(vm: TrackerViewModel = viewModel()) {
    val stats by vm.statsFlow.collectAsState()
    val words by vm.wordsFlow.collectAsState()
    val units by vm.unitsFlow.collectAsState()
    val summaries by vm.levelSummariesFlow.collectAsState()
    val medals by vm.medalsFlow.collectAsState()
    val isTimerRunning by vm.isTimerRunning.collectAsState()
    val sessionSeconds by vm.sessionSeconds.collectAsState()

    val currentDialect = remember(stats.selectedDialect) {
        runCatching { EnglishDialect.valueOf(stats.selectedDialect) }.getOrDefault(EnglishDialect.US)
    }

    var selectedTab by remember { mutableStateOf(NavTab.DASHBOARD) }
    var selectedLevelFilter by remember { mutableStateOf<CefrLevel?>(null) }

    // Инициализация синтезатора речи (TTS) с поддержкой US и UK произношения
    val context = LocalContext.current
    var ttsEngine by remember { mutableStateOf<TextToSpeech?>(null) }
    DisposableEffect(context) {
        var instance: TextToSpeech? = null
        instance = TextToSpeech(context) { status ->
            if (status == TextToSpeech.SUCCESS) {
                instance?.language = Locale.US
            }
        }
        ttsEngine = instance
        onDispose {
            instance?.stop()
            instance?.shutdown()
        }
    }

    val speakWord: (String, EnglishDialect) -> Unit = { text, dialect ->
        ttsEngine?.let { tts ->
            tts.language = if (dialect == EnglishDialect.US) Locale.US else Locale.UK
            tts.speak(text, TextToSpeech.QUEUE_FLUSH, null, "oxford_word_tts")
        }
    }

    Scaffold(
        containerColor = SoftPaperColors.Background,
        topBar = {
            MinimalistTopBar(
                dialect = currentDialect,
                streakDays = stats.currentStreakDays,
                onDialectToggle = {
                    val next = if (currentDialect == EnglishDialect.US) EnglishDialect.UK else EnglishDialect.US
                    vm.setDialect(next)
                }
            )
        },
        bottomBar = {
            NavigationBar(
                containerColor = SoftPaperColors.SurfaceCard,
                tonalElevation = 0.dp
            ) {
                NavTab.entries.forEach { tab ->
                    NavigationBarItem(
                        selected = selectedTab == tab,
                        onClick = { selectedTab = tab },
                        icon = { Icon(tab.icon, contentDescription = tab.label) },
                        label = { Text(tab.label, fontSize = 11.sp, fontWeight = FontWeight.Medium) },
                        colors = NavigationBarItemDefaults.colors(
                            selectedIconColor = SoftPaperColors.SagePrimary,
                            selectedTextColor = SoftPaperColors.SagePrimary,
                            indicatorColor = SoftPaperColors.SageLight,
                            unselectedIconColor = SoftPaperColors.TextSecondary,
                            unselectedTextColor = SoftPaperColors.TextSecondary
                        )
                    )
                }
            }
        }
    ) { innerPadding ->
        Box(
            modifier = Modifier
                .fillMaxSize()
                .padding(innerPadding)
        ) {
            when (selectedTab) {
                NavTab.DASHBOARD -> DashboardScreen(
                    stats = stats,
                    summaries = summaries,
                    isTimerRunning = isTimerRunning,
                    sessionSeconds = sessionSeconds,
                    onToggleTimer = vm::toggleTimer,
                    onFinishTimer = vm::finishAndSaveTimerSession,
                    onQuickAddMinutes = vm::addStudyMinutes,
                    onOpenLevelAnki = { level ->
                        selectedLevelFilter = level
                        selectedTab = NavTab.ANKI
                    },
                    onOpenLevelMurphy = { level ->
                        selectedLevelFilter = level
                        selectedTab = NavTab.MURPHY
                    }
                )
                NavTab.ANKI -> AnkiDeckScreen(
                    words = words,
                    summaries = summaries,
                    dialect = currentDialect,
                    selectedLevel = selectedLevelFilter,
                    onSelectLevel = { selectedLevelFilter = it },
                    onGradeWord = vm::submitAnkiReview,
                    onSpeak = { text -> speakWord(text, currentDialect) }
                )
                NavTab.OXFORD -> OxfordDictionaryScreen(
                    words = words,
                    summaries = summaries,
                    dialect = currentDialect,
                    selectedLevel = selectedLevelFilter,
                    onSelectLevel = { selectedLevelFilter = it },
                    onSpeak = { text -> speakWord(text, currentDialect) },
                    onAddWord = vm::addCustomOxfordWord
                )
                NavTab.MURPHY -> MurphyGrammarScreen(
                    units = units,
                    summaries = summaries,
                    selectedLevel = selectedLevelFilter,
                    onSelectLevel = { selectedLevelFilter = it },
                    onToggleUnit = vm::toggleMurphyUnit
                )
                NavTab.MEDALS -> MedalsScreen(
                    medals = medals,
                    stats = stats,
                    summaries = summaries
                )
            }
        }
    }
}

// ЭКРАН 2: КАРТОЧКИ ANKI (ТРАНСКРИПЦИЯ US/UK + ПЕРЕВОД НА РУССКИЙ ПОСЛЕ ОТВЕТА)

@Composable
fun AnkiDeckScreen(
    words: List<OxfordWordEntity>,
    summaries: List<LevelProgressSummary>,
    dialect: EnglishDialect,
    selectedLevel: CefrLevel?,
    onSelectLevel: (CefrLevel?) -> Unit,
    onGradeWord: (OxfordWordEntity, AnkiGrade) -> Unit,
    onSpeak: (String) -> Unit
) {
    // Фильтруем карточки по выбранному уровню; в первую очередь показываем слова, которые еще не закрыты идеально (perfectStreak < 3)
    val deckWords = remember(words, selectedLevel) {
        val filtered = if (selectedLevel == null) words else words.filter { it.levelCode == selectedLevel.code }
        filtered.sortedWith(
            compareBy<OxfordWordEntity> { it.isMasteredInAnki }
                .thenBy { it.nextReviewEpochDay }
                .thenBy { it.perfectStreak }
        )
    }

    var currentIndex by remember(selectedLevel, deckWords.size) { mutableIntStateOf(0) }
    // Состояние карточки: null = пользователь еще не ответил; иначе содержит выбранную оценку и открывает перевод на русский
    var answeredGrade by remember { mutableStateOf<AnkiGrade?>(null) }

    val currentWord = deckWords.getOrNull(currentIndex % max(1, deckWords.size))

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(14.dp)
    ) {
        // Фильтр по уровням CEFR
        LevelFilterBar(
            selectedLevel = selectedLevel,
            onSelectLevel = {
                onSelectLevel(it)
                answeredGrade = null
                currentIndex = 0
            }
        )

        if (currentWord == null) {
            Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                Text("Нет карточек для выбранного уровня", color = SoftPaperColors.TextSecondary)
            }
            return@Column
        }

        // Статус закрытия словаря выбранного уровня
        val unmasteredCount = deckWords.count { !it.isMasteredInAnki }
        Surface(
            shape = RoundedCornerShape(12.dp),
            color = SoftPaperColors.SurfaceSubtle
        ) {
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 14.dp, vertical = 10.dp),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = "Осталось идеально сдать до закрытия:",
                    fontSize = 12.sp,
                    color = SoftPaperColors.TextSecondary
                )
                Text(
                    text = "$unmasteredCount из ${deckWords.size} слов",
                    fontSize = 13.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = SoftPaperColors.SagePrimary
                )
            }
        }

        // Главная карточка Anki
        Surface(
            shape = RoundedCornerShape(20.dp),
            color = SoftPaperColors.SurfaceCard,
            border = BorderStroke(1.dp, SoftPaperColors.BorderWarm),
            modifier = Modifier
                .fillMaxWidth()
                .weight(1f)
        ) {
            Column(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(24.dp),
                horizontalAlignment = Alignment.CenterHorizontally,
                verticalArrangement = Arrangement.SpaceBetween
            ) {
                // Верхняя строка карточки: уровень, часть речи и серия идеальных ответов (X/3)
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically

@Composable
fun AddOxfordWordDialog(
    initialLevel: CefrLevel,
    onDismiss: () -> Unit,
    onSave: (CefrLevel, String, String, String, String, String, String, String, String) -> Unit
) {
    var level by remember { mutableStateOf(initialLevel) }
    var wordUs by remember { mutableStateOf("") }
    var wordUk by remember { mutableStateOf("") }
    var ipaUs by remember { mutableStateOf("") }
    var translationRu by remember { mutableStateOf("") }
    var exampleEn by remember { mutableStateOf("") }
    var exampleRu by remember { mutableStateOf("") }

    AlertDialog(
        onDismissRequest = onDismiss,
        containerColor = SoftPaperColors.SurfaceCard,
        title = { Text("Добавить слово в базу Oxford", fontSize = 17.sp, fontWeight = FontWeight.Bold) },
        text = {
            Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
                OutlinedTextField(
                    value = wordUs,
                    onValueChange = { wordUs = it },
                    label = { Text("Слово (US English)") },
                    singleLine = true
                )
                OutlinedTextField(
                    value = wordUk,
                    onValueChange = { wordUk = it },
                    label = { Text("Британский вариант (UK, если отличается)") },
                    singleLine = true
                )
                OutlinedTextField(
                    value = ipaUs,
                    onValueChange = { ipaUs = it },
                    label = { Text("Транскрипция (например /ˈwɝːd/)") },
                    singleLine = true
                )
                OutlinedTextField(
                    value = translationRu,
                    onValueChange = { translationRu = it },
                    label = { Text("Перевод на русский") },
                    singleLine = true
                )
                OutlinedTextField(
                    value = exampleEn,
                    onValueChange = { exampleEn = it },
                    label = { Text("Пример предложения (EN)") },
                    singleLine = true
                )
            }
        },
        confirmButton = {
            Button(
                onClick = {
                    if (wordUs.isNotBlank() && translationRu.isNotBlank()) {
                        onSave(level, wordUs, wordUk, ipaUs, ipaUs, "word", translationRu, exampleEn, exampleRu)
                    }
                },
                colors = ButtonDefaults.buttonColors(containerColor = SoftPaperColors.SagePrimary)
            ) {
                Text("Добавить")
            }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) {
                Text("Отмена")
            }
        }
    )
}

                            trackColor = SoftPaperColors.SurfaceSubtle
                        )
                        Spacer(modifier = Modifier.width(12.dp))
                        Text(
                            text = medal.progressText,
                            fontSize = 11.sp,
                            fontWeight = FontWeight.SemiBold,
                            color = SoftPaperColors.TextSecondary
                        )
                    }
                }
            }
        }
    }
}

@Composable
fun LevelFilterBar(
    selectedLevel: CefrLevel?,
    onSelectLevel: (CefrLevel?) -> Unit
) {
    LazyRow(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
        item {
            FilterChip(
                selected = selectedLevel == null,
                onClick = { onSelectLevel(null) },
                label = { Text("Все уровни") }
            )
        }
        items(CefrLevel.entries) { level ->
            FilterChip(
                selected = selectedLevel == level,
                onClick = { onSelectLevel(level) },
                label = { Text(level.code) }
            )
        }
    }
}
