fun parseCsv(text: String): List<List<String>> {
    val rows = ArrayList<List<String>>()
    var row = ArrayList<String>()
    val sb = StringBuilder()
    var q = false
    val t = text.removePrefix("\uFEFF")
    var i = 0
    while (i < t.length) {
        val c = t[i]
        if (q) {
            if (c == '"') {
                if (i + 1 < t.length && t[i + 1] == '"') {
                    sb.append('"')
                    i++
                } else {
                    q = false
                }
            } else {
                sb.append(c)
            }
        } else if (c == '"') {
            q = true
        } else if (c == ';') {
            row.add(sb.toString())
            sb.setLength(0)
        } else if (c == '\n') {
            row.add(sb.toString())
            sb.setLength(0)
            rows.add(row)
            row = ArrayList()
        } else if (c != '\r') {
            sb.append(c)
        }
        i++
    }
    if (sb.isNotEmpty() || row.isNotEmpty()) {
        row.add(sb.toString())
        rows.add(row)
    }
    return rows
}

fun parseWords(text: String): List<Word> {
    val out = ArrayList<Word>()
    for (r in parseCsv(text)) {
        if (r.size < 9) continue
        val level = r[1].trim().uppercase()
        val w = r[2].trim()
        if (w.isEmpty() || level !in LEVELS) continue
        val tr = r[6].split("|").map { it.trim() }.filter { it.isNotEmpty() }
        if (tr.isEmpty()) continue
        out.add(
            Word(
                w.lowercase(), level, w, r[3].trim(), r[4].trim(), r[5].trim(), tr,
                r[8].trim(), r.getOrElse(9) { "" }.trim(), r.getOrElse(10) { "" }.trim()
            )
        )
    }
    return out
}

fun norm(s: String): String =
    s.lowercase().replace('ё', 'е').filter { it.isLetterOrDigit() || it == ' ' }.trim().replace(Regex("\\s+"), " ")

fun lev(a: String, b: String): Int {
    if (a == b) return 0
    var prev = IntArray(b.length + 1) { it }
    for (i in 1..a.length) {
        val cur = IntArray(b.length + 1)
        cur[0] = i
        for (j in 1..b.length) {
            val cost = if (a[i - 1] == b[j - 1]) 0 else 1
            cur[j] = minOf(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + cost)
        }
        prev = cur
    }
    return prev[b.length]
}

fun wordAt(s: String, off: Int): String? {
    if (s.isEmpty()) return null
    var a = off.coerceIn(0, s.length - 1)
    var b = a
    if (!(s[a].isLetter() || s[a] == '\'')) return null
    while (a > 0 && (s[a - 1].isLetter() || s[a - 1] == '\'')) a--
    while (b < s.length - 1 && (s[b + 1].isLetter() || s[b + 1] == '\'')) b++
    return s.substring(a, b + 1).lowercase()
}

    fun today(): Long = LocalDate.now().toEpochDay()

    private fun rebuild() {
        val m = LinkedHashMap<String, Word>()
        for (w in baseWords + custom) m.putIfAbsent(w.key, w)
        words = m.values.sortedBy { LEVELS.indexOf(it.level) }
    }

    // ---------- сохранение ----------

    private fun toJson(): JSONObject {
        val o = JSONObject()
        o.put("level", level)
        o.put("startDone", startDone)
        o.put("limit", limit)
        o.put("accent", accent)
        o.put("nextTest", nextTest)
        o.put("newPaused", newPaused)
        val p = JSONObject()
        for ((k, v) in prog) {
            p.put(k, JSONArray(listOf(v.s, v.due, if (v.mastered) 1 else 0, v.intro, v.lapses, v.lastWrong)))
        }
        o.put("prog", p)
        val d = JSONObject()
        for ((k, v) in days) d.put(k.toString(), v)
        o.put("days", d)
        o.put("unitsPassed", JSONArray(unitPassed.filter { it.value }.keys.toList()))
        o.put("unitsReview", JSONArray(unitReview.filter { it.value }.keys.toList()))
        o.put("tests", JSONArray(tests.map { JSONArray(listOf(it.day, it.kind, it.level, it.percent, if (it.passed) 1 else 0)) }))
        o.put("custom", JSONArray(custom.map { JSONArray(listOf(it.word, it.level, it.tr.joinToString("|"), it.example, it.exampleRu)) }))
        o.put("pinned", JSONArray(pinned.toList()))
        return o
    }

    private fun load(o: JSONObject) {

        startDone = o.optBoolean("startDone", false)
        limit = o.optInt("limit", 10)
        accent = o.optString("accent", "UK")
        nextTest = o.optLong("nextTest", 0L)
        newPaused = o.optBoolean("newPaused", false)
        prog.clear(); days.clear(); unitPassed.clear(); unitReview.clear()
        tests.clear(); custom.clear(); pinned.clear()
        o.optJSONObject("prog")?.let { p ->
            for (k in p.keys()) {
                val a = p.getJSONArray(k)
                prog[k] = Prog(a.getInt(0), a.getLong(1), a.getInt(2) == 1, a.getLong(3), a.getInt(4), a.getLong(5))
            }
        }
        o.optJSONObject("days")?.let { d ->
            for (k in d.keys()) days[k.toLong()] = d.getInt(k)
        }
        o.optJSONArray("unitsPassed")?.let { a ->
            for (i in 0 until a.length()) unitPassed[a.getString(i)] = true
        }
        o.optJSONArray("unitsReview")?.let { a ->
            for (i in 0 until a.length()) unitReview[a.getString(i)] = true
        }
        o.optJSONArray("tests")?.let { a ->
            for (i in 0 until a.length()) {

        }
        o.optJSONArray("pinned")?.let { a ->
            for (i in 0 until a.length()) pinned.add(a.getString(i))
        }
    }

    // ---------- время и стрик ----------

    fun touch() {
        lastTouch = System.currentTimeMillis()
    }

    fun tick(count: Boolean) {
        if (!count || System.currentTimeMillis() - lastTouch > 120_000) return
        pending++
        if (pending >= 10) flush()
    }

    fun flush() {
        if (pending > 0) {
            val d = today()
            days[d] = (days[d] ?: 0) + pending
            pending = 0
            save()
        }
    }

    fun totalSec(): Int = days.values.sum() + pending
    fun hoursLabel(): String = fmt1(totalSec() / 3600f)

    private fun testDays(): Set<Long> = tests.filter { it.passed }.map { it.day }.toSet()
    private fun counted(d: Long, td: Set<Long>): Boolean = (days[d] ?: 0) >= 300 || d in td

    fun streak(): Int {
        val td = testDays()
        var d = today()
        if (!counted(d, td)) d--
        var n = 0
        while (counted(d, td)) {
            n++
            d--
        }
        return n
    }

    fun record(): Int {
        val td = testDays()
        val ds = (days.filter { it.value >= 300 }.keys + td).toSortedSet()
        var best = 0
        var run = 0
        var prev = Long.MIN_VALUE
        for (d in ds) {
            run = if (d == prev + 1) run + 1 else 1
            if (run > best) best = run
            prev = d
        }
        return best
    }

    // ---------- Anki ----------

    fun dueWords(filter: String?): List<Word> {
        val t = today()
        return words.filter { w ->
            (filter == null || w.level == filter) && prog[w.key].let { it != null && !it.mastered && it.due <= t }
        }.sortedBy { prog[it.key]?.due ?: 0L }
    }

    fun introducedToday(): Int {
        val t = today()
        return prog.values.count { it.intro == t }
    }

    fun newLeft(): Int = if (newPaused) 0 else maxOf(0, limit - introducedToday())

    private fun nextNew(filter: String?): Word? {
        for (k in pinned) {
            val w = words.firstOrNull { it.key == k }
            if (w != null && prog[k] == null && (filter == null || w.level == filter)) return w
        }
        return words.firstOrNull { (filter == null || it.level == filter) && prog[it.key] == null }
    }

    fun pickCard(filter: String?): Word? {
        val d = dueWords(filter).firstOrNull()
        if (d != null) return d
        return if (newLeft() > 0) nextNew(filter) else null
    }

    private fun updatePause() {
        val n = dueWords(null).size
        if (n > 50) newPaused = true else if (n < 20) newPaused = false
    }

    private fun checkAnswer(w: Word, input: String): Pair<Boolean, String?> {
        val x = norm(input)
        if (x.isEmpty()) return Pair(false, null)
        var bestD = Int.MAX_VALUE
        var bestT = ""
        for (t in w.tr) {
            val d = lev(x, norm(t))
            if (d < bestD) {
                bestD = d
                bestT = t
            }
        }
        val len = norm(bestT).length
        val allow = if (len >= 8) 2 else if (len >= 5) 1 else 0
        return if (bestD == 0) Pair(true, null) else if (bestD <= allow) Pair(true, bestT) else Pair(false, null)
    }

    private fun applyResult(key: String, ok: Boolean): String {
        val t = today()
        val old = prog[key]
        val intro = old?.intro ?: t
        val lapses = old?.lapses ?: 0
        val lastWrong = old?.lastWrong ?: 0L
        pinned.remove(key)
        val info: String
        if (ok) {
            val s = (old?.s ?: 0) + 1
            if (s >= 6) {
                prog[key] = Prog(6, t, true, intro, lapses, lastWrong)
                info = "Слово закреплено"
            } else {
                prog[key] = Prog(s, t + INTERVALS[s - 1], false, intro, lapses, lastWrong)
                info = "Повторим через ${INTERVALS[s - 1]} дн."
            }
        } else {
            prog[key] = Prog(0, t + 1, false, intro, lapses + 1, t)
            info = "Повторим завтра"
        }
        updatePause()
        save()
        return info
    }

    fun answer(w: Word, input: String?): Res {
        var ok = false
        var fixed: String? = null
        if (input != null) {
            val c = checkAnswer(w, input)
            ok = c.first
            fixed = c.second
        }
        val info = applyResult(w.key, ok)
        return Res(ok, fixed, info)
    }

    fun pin(key: String) {
        if (key !in pinned && prog[key] == null) pinned.add(key)
        save()
    }

    fun addCustom(word: String, tr: String, example: String): Boolean {
        val w = word.trim()
        val t = tr.split("|", ",").map { it.trim() }.filter { it.isNotEmpty() }
        if (w.isEmpty() || t.isEmpty()) return false
        custom.add(Word(w.lowercase(), level, w, "", "", "", t, example.trim(), "", ""))
        rebuild()
        pin(w.lowercase())
        return true
    }

    // ---------- грамматика ----------

    fun finishUnit(id: String, ok: Boolean) {
        if (ok) {
            unitPassed[id] = true
            unitReview.remove(id)
        } else {
            unitReview[id] = true
        }
        save()
    }

    // ---------- тесты ----------

    private fun grammarQs(l: String): List<Q> =
        UNITS.filter { it.level == l }.flatMap { u ->
            u.quiz.map { q -> Q("grammar", l, q.text, q.options, q.answer, "Правило: ${u.formula}", null, u.id) }
        }

    private fun readingQs(l: String): List<Q> =
        TEXTS.filter { it.level == l }.flatMap { t ->
            t.questions.map { q -> Q("reading", l, q.text, q.options, q.answer, "Ответ есть в тексте «${t.title}»", null, null, t.body) }
        }

    private fun vocabQs(l: String, n: Int): List<Q> {
        if (n <= 0) return emptyList()
        val out = ArrayList<Q>()
        for (w in words.filter { it.level == l }.shuffled().take(n)) {
            val right = w.tr.first()
            val near = words.filter { it.level == l && it.key != w.key }.map { it.tr.first() }
                .distinct().filter { it != right }.shuffled()
            val far = words.filter { it.key != w.key }.map { it.tr.first() }
                .distinct().filter { it != right && it !in near }.shuffled()
            val wrong = (near + far).take(3)
            if (wrong.size < 3) continue
            val opts = (wrong + right).shuffled()
            out.add(
                Q("vocab", l, "Что значит слово «${w.word}»?", opts, opts.indexOf(right),
                    "${w.word} — ${w.tr.joinToString(", ")}", w.key, null)
            )
        }
        return out
    }

    fun startStartTest() {
        val qs = ArrayList<Q>()
        for (l in LEVELS.drop(1)) {
            val g = grammarQs(l).shuffled().take(3)
            qs.addAll(g)
            qs.addAll(vocabQs(l, 5 - g.size))
        }
        if (qs.isNotEmpty()) test = TestSession("start", "A1", qs, 0L)
    }

    fun weeklyTarget(): String = LEVELS.getOrElse(LEVELS.indexOf(level) + 1) { "C2" }
    fun weeklyAvailable(): Boolean = startDone && today() >= nextTest

    fun startWeekly() {
        val l = weeklyTarget()
        val qs = grammarQs(l).shuffled().take(20) + vocabQs(l, 25) + readingQs(l)
        if (qs.isNotEmpty()) test = TestSession("weekly", l, qs, 40L * 60L * 1000L)
    }

    fun nextQ() {
        val s = test ?: return
        if (s.result != null) return
        val cur = s.qs[s.index]
        if (s.index + 1 >= s.qs.size) {
            finishTest()
            return
        }
        if (s.kind == "start" && s.qs[s.index + 1].level != cur.level) {
            var a = s.index
            while (a > 0 && s.qs[a - 1].level == cur.level) a--
            var right = 0
            for (k in a..s.index) if (s.answers[k] == s.qs[k].answer) right++
            if (right * 5 < (s.index - a + 1) * 4) {
                finishTest()
                return
            }
        }
        s.index += 1
    }

    fun finishTest() {
        val s = test ?: return
        if (s.result != null) return
        val last = if (s.kind == "start") s.index else s.qs.lastIndex
        val wrong = ArrayList<Pair<Q, Int>>()
        val secTotal = HashMap<String, Int>()
        val secRight = HashMap<String, Int>()
        var right = 0
        for (k in 0..last) {
            val q = s.qs[k]
            secTotal[q.section] = (secTotal[q.section] ?: 0) + 1
            if (s.answers[k] == q.answer) {
                right++
                secRight[q.section] = (secRight[q.section] ?: 0) + 1
            } else {
                wrong.add(Pair(q, s.answers[k]))
            }
        }
        val count = last + 1
        val percent = if (count <= 0) 0 else right * 100 / count
        val secs = listOf("grammar", "vocab", "reading").filter { secTotal.containsKey(it) }
            .map { sec -> Pair(sec, (secRight[sec] ?: 0) * 100 / (secTotal[sec] ?: 1)) }
        val passed: Boolean
        var newLevel: String? = null
        val t = today()
        if (s.kind == "start") {
            var best = "A0"
            for (l in LEVELS.drop(1)) {
                val ks = (0..last).filter { s.qs[it].level == l }
                if (ks.isEmpty()) break
                val r = ks.count { s.answers[it] == s.qs[it].answer }
                if (r * 5 >= ks.size * 4) best = l else break
            }
            level = best
            startDone = true
            passed = true
            newLevel = best
        } else {
            passed = percent >= 75 && secs.all { it.second >= 60 }
            if (passed) {
                level = s.target
                newLevel = s.target
            }
        }
        nextTest = t + 7
        tests.add(TestRec(t, s.kind, s.target, percent, passed))
        for (pair in wrong) {
            val q = pair.first
            val wk = q.wordKey
            if (wk != null) applyResult(wk, false)
            val uid = q.unitId
            if (uid != null) unitReview[uid] = true
        }
        s.result = TestResult(percent, passed, newLevel, secs, wrong)
        updatePause()
        save()
    }

    // ---------- прогресс и медали ----------

    fun levelProgress(l: String): Float {
        val ws = words.filter { it.level == l }
        val us = UNITS.filter { it.level == l }
        val parts = ArrayList<Float>()
        if (ws.isNotEmpty()) parts.add(ws.count { prog[it.key]?.mastered == true }.toFloat() / ws.size)
        if (us.isNotEmpty()) parts.add(us.count { unitPassed[it.id] == true }.toFloat() / us.size)
        return if (parts.isEmpty()) 0f else parts.sum() / parts.size
    }

    fun nextLevel(): String = LEVELS.getOrElse(LEVELS.indexOf(level) + 1) { "C2" }

    fun nextProgressLabel(): String = "К ${nextLevel()}: ${(levelProgress(nextLevel()) * 100).toInt()}%"

    fun nextTestLabel(): String = if (!startDone) "стартовый" else LocalDate.ofEpochDay(nextTest).toString()

    fun weakItems(): List<String> {
        val us = UNITS.filter { unitReview[it.id] == true }.map { it.title }
        val ws = prog.filter { it.value.lapses > 0 && !it.value.mastered && it.value.s <= 1 }
            .entries.sortedByDescending { it.value.lastWrong }.map { it.key }
        return (us + ws).take(10)
    }

    fun streakMedals(): List<Medal> {
        val r = record()
        return listOf(7, 14, 21, 28, 60, 90, 120, 150, 180, 210, 240, 270, 300, 330, 360).map { n ->
            Medal("$n ${daysWord(n)} подряд", "Занимайся $n ${daysWord(n)} без пропусков", r >= n,
                r.toFloat() / n, "${minOf(r, n)} / $n дн.")
        }
    }

    fun timeMedals(): List<Medal> {
        val h = totalSec() / 3600f
        return listOf(1, 10, 20, 50, 100, 200, 500, 1000).map { n ->
            Medal(if (n == 1) "Новичок: 1 час" else "$n часов практики", "Набери $n ч занятий в приложении",
                h >= n, h / n, "${fmt1(minOf(h, n.toFloat()))} / $n ч")
        }
    }

    fun levelMedals(): List<Medal> = LEVELS.drop(1).map { l ->
        val done = LEVELS.indexOf(level) >= LEVELS.indexOf(l)
        Medal("Уровень $l", "Пройди тест на уровень $l", done, if (done) 1f else 0f, if (done) "Получено" else "Не получено")
    }
}

@Composable
fun EnglishTrackerRoot(vm: AppVm, resumed: Boolean) {
    val ctx = LocalContext.current
    var tab by rememberSaveable { mutableStateOf(0) }
    val tts = remember { TextToSpeech(ctx, null) }
    DisposableEffect(Unit) { onDispose { tts.shutdown() } }
    val speak: (String, Boolean) -> Unit = { text, us ->
        tts.setLanguage(if (us) Locale.US else Locale.UK)
        tts.speak(text, TextToSpeech.QUEUE_FLUSH, null, "w")
        vm.touch()
    }
    // Время считается само: пока приложение открыто, на экране, и были касания в последние 2 минуты.
    LaunchedEffect(resumed, tab) {
        while (resumed) {
            delay(1000)
            vm.tick(tab != 4)
        }
    }
    BackHandler(enabled = vm.test != null) { vm.test = null }
    Column(
        Modifier.fillMaxSize().background(BgC).pointerInput(Unit) {
            awaitPointerEventScope {
                while (true) {
                    awaitPointerEvent(PointerEventPass.Initial)
                    vm.touch()
                }
            }
        }
    ) {
        AppTopBar(vm)
        Box(Modifier.weight(1f).fillMaxWidth()) {
            val s = vm.test
            if (s != null) {
                TestBody(vm, s)
            } else {
                when (tab) {
                    0 -> ProgressScreen(vm)
                    1 -> AnkiScreen(vm, speak)
                    2 -> DictScreen(vm, speak)
                    3 -> GrammarScreen(vm)
                    else -> ProfileScreen(vm)
                }
            }
        }
        if (vm.test == null) BottomBar(tab) { tab = it }
    }
}

@Composable
fun DictScreen(vm: AppVm, speak: (String, Boolean) -> Unit) {
    val ctx = LocalContext.current
    var q by remember { mutableStateOf("") }
    var filter by remember { mutableStateOf<String?>(null) }
    var showAdd by remember { mutableStateOf(false) }
    val picker = rememberLauncherForActivityResult(ActivityResultContracts.OpenDocument()) { uri: Uri? ->
        if (uri != null) {
            val text = runCatching {
                ctx.contentResolver.openInputStream(uri)?.bufferedReader(Charsets.UTF_8)?.use { it.readText() }
            }.getOrNull()
            val n = if (text == null) 0 else vm.importCsv(text)
            ctx.toast(if (n > 0) "Загружено слов: $n" else "Не удалось прочитать словарь")
        }
    }
    val list = remember(vm.words, q, filter) {
        val s = q.trim().lowercase()
        vm.words.filter { w ->
            (filter == null || w.level == filter) &&
                (s.isEmpty() || w.key.contains(s) || w.tr.any { t -> t.lowercase().contains(s) })
        }
    }
    LazyColumn(
        Modifier.fillMaxSize(), contentPadding = PaddingValues(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        item {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Column(Modifier.weight(1f)) {
                    Text("Словарь", fontSize = 24.sp, fontWeight = FontWeight.Bold, color = InkC)
                    Sub("Слов: ${vm.words.size}  •  A0–C2")
                }
                IconButton(onClick = { showAdd = true }) {
                    Icon(Icons.Filled.Add, contentDescription = "Добавить слово", tint = GreenC)
                }
            }
        }
        item {
            Column {
                Secondary("Загрузить словарь (CSV)", Modifier.fillMaxWidth()) { picker.launch(arrayOf("*/*")) }
                Gap(6)
                Sub("Колонки через «;»: id; level; word; pos; ipa; us_word; translations_ru; translations_en; example; example_tr; emoji. Варианты перевода — через «|».", size = 12)
            }
        }
        item {
            OutlinedTextField(
                value = q, onValueChange = { q = it }, singleLine = true, label = { Text("Поиск") },
                modifier = Modifier.fillMaxWidth(), shape = RoundedCornerShape(16.dp)
            )
        }
        item { LevelFilter(filter) { filter = it } }
        items(list, key = { it.key }) { w -> WordRow(vm, w, speak) }
    }
    if (showAdd) AddWordDialog(vm, "") { showAdd = false }
}

@Composable
fun UnitScreen(vm: AppVm, u: GUnit, onBack: () -> Unit) {
    val picks = remember(u.id) { mutableStateListOf<Int>().also { l -> repeat(u.quiz.size) { l.add(-1) } } }
    var checked by remember(u.id) { mutableStateOf(false) }
    BackHandler(onBack = onBack)
    Column(
        Modifier.fillMaxSize().verticalScroll(rememberScrollState()).padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        TextButton(onClick = onBack) { Text("← Назад", color = GreenC) }
        Text(u.title, fontSize = 24.sp, fontWeight = FontWeight.Bold, color = InkC)
        AppCard {
            H("Правило")
            Gap(6)
            Text(u.rule, fontSize = 16.sp, color = InkC, lineHeight = 24.sp)
        }
        AppCard {
            H("Схема")
            Gap(8)
            Box(Modifier.fillMaxWidth().clip(RoundedCornerShape(14.dp)).background(GreenSoftC).padding(14.dp)) {
                Text(u.formula, fontSize = 16.sp, fontWeight = FontWeight.SemiBold, color = GreenC)
            }
        }
        AppCard {
            H("Примеры")
            Gap(6)
            u.examples.forEach { ex ->
                Text(ex, fontSize = 15.sp, color = InkC, lineHeight = 22.sp)
                Gap(6)
            }
        }
        AppCard {
            H("Частая ошибка", 18, RedC)
            Gap(6)
            Text(u.mistake, fontSize = 15.sp, color = InkC, lineHeight = 22.sp)
        }
        H("Мини-тест", 20)
        u.quiz.forEachIndexed { qi, q ->
            AppCard {
                Text("${qi + 1}. ${q.text}", fontSize = 17.sp, fontWeight = FontWeight.SemiBold, color = InkC)
                q.options.forEachIndexed { oi, o ->
                    val sel = picks[qi] == oi
                    val bg = when {
                        !checked -> if (sel) GreenSoftC else ChipC
                        oi == q.answer -> GreenSoftC
                        sel -> RedSoftC
                        else -> ChipC
                    }
                    OptionRow(o, bg, sel && !checked, !checked) { picks[qi] = oi }
                }
            }
        }
        val score = u.quiz.indices.count { picks[it] == u.quiz[it].answer }
        if (!checked) {
            Primary("Проверить", Modifier.fillMaxWidth(), picks.all { it >= 0 }) {
                checked = true
                vm.finishUnit(u.id, score >= 4)
            }
        } else {
            AppCard {
                H("Результат: $score из ${u.quiz.size}")
                Gap(4)
                Sub(if (score >= 4) "Юнит пройден." else "Нужно минимум 4. Перечитай правило и попробуй ещё раз.")
            }
            Secondary("Пройти ещё раз", Modifier.fillMaxWidth()) {
                for (i in picks.indices) picks[i] = -1
                checked = false
            }
        }
        Gap(16)
    }
}

@Composable
fun TestBody(vm: AppVm, s: TestSession) {
    var now by remember(s) { mutableStateOf(System.currentTimeMillis()) }
    LaunchedEffect(s) {
        while (s.result == null) {
            delay(1000)
            now = System.currentTimeMillis()
            if (s.limitMs > 0 && now - s.startedAt >= s.limitMs) vm.finishTest()
        }
    }
    val r = s.result
    if (r != null) {
        ResultView(vm, s, r)
    } else {
        val q = s.qs[s.index]
        val left: Long = if (s.limitMs > 0) maxOf(0L, s.limitMs - (now - s.startedAt)) / 1000 else -1L
        Column(
            Modifier.fillMaxSize().verticalScroll(rememberScrollState()).padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(
                    if (s.kind == "start") "Стартовый тест" else "Тест недели • ${s.target}",
                    fontSize = 18.sp, fontWeight = FontWeight.Bold, color = InkC, modifier = Modifier.weight(1f)
                )
                if (left >= 0) {
                    Text(
                        "%02d:%02d".format(left / 60, left % 60), fontSize = 20.sp, fontWeight = FontWeight.Bold,
                        color = if (left < 300) RedC else GreenC
                    )
                }
            }
            Sub("Вопрос ${s.index + 1} из ${s.qs.size}  •  ${sectionName(q.section)}")
            Bar((s.index + 1).toFloat() / s.qs.size)
            val ps = q.passage
            if (ps != null) {
                AppCard { Text(ps, fontSize = 16.sp, color = InkC, lineHeight = 24.sp) }
            }
            AppCard {
                Text(q.text, fontSize = 20.sp, fontWeight = FontWeight.SemiBold, color = InkC)
                q.options.forEachIndexed { i, o ->
                    val sel = s.answers[s.index] == i
                    OptionRow(o, if (sel) GreenSoftC else ChipC, sel) {
                        vm.touch()
                        s.answers[s.index] = i
                    }
                }
            }
            Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                Secondary("Прервать", Modifier.weight(1f)) { vm.test = null }
                Primary(
                    if (s.index + 1 >= s.qs.size) "Завершить" else "Далее", Modifier.weight(1f),
                    s.answers[s.index] >= 0
                ) { vm.nextQ() }
            }
            Gap(16)
        }
    }
}

@Composable

    Column(
        Modifier.fillMaxSize().verticalScroll(rememberScrollState()).padding(16.dp),
        verticalArrangement = Arrangement.spacedBy(12.dp)
    ) {
        AppCard {
            val title = if (r.passed) {
                if (s.kind == "start") "Уровень определён" else "Тест пройден"
            } else "Тест не пройден"
            H(title, 22, if (r.passed) GreenC else RedC)
            Text("${r.percent}%", fontSize = 48.sp, fontWeight = FontWeight.Bold, color = InkC)
            r.sections.forEach { sec -> Sub("${sectionName(sec.first)}: ${sec.second}%", size = 15) }
            if (r.newLevel != null) {
                Gap(8)
                H("Твой уровень: ${r.newLevel}", 18, GreenC)
            }
            if (s.kind == "weekly" && !r.passed) {
                Gap(8)
                Sub("Нужно 75% в целом и 60% в каждом разделе. Следующая попытка через неделю.")
            }
        }
        if (r.wrong.isNotEmpty()) {
            H("Разбор ошибок", 20)
            r.wrong.forEach { pair ->
                val q = pair.first
                val a = pair.second
                AppCard {
                    Text(q.text, fontSize = 16.sp, fontWeight = FontWeight.SemiBold, color = InkC)

                    Text("Верно: ${q.options[q.answer]}", fontSize = 15.sp, color = GreenC)
                    Gap(4)
                    Sub(q.tip, size = 13)
                }
            }
        }
        Primary("Готово", Modifier.fillMaxWidth()) { vm.test = null }
        Gap(16)
    }
}
